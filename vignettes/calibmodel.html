Make predictions on test set

lm_pred <- predict(lm_model, newdata = test_data, interval = "prediction")
calib_pred <- predict(calib_model, newdata = as.matrix(test_data[,-1]), interval = "prediction")

Compare predictions

results <- data.frame(
  Actual = test_data$mpg,
  LM_Pred = lm_pred[,"fit"],
  LM_Lower = lm_pred[,"lwr"],
  LM_Upper = lm_pred[,"upr"],
  Calib_Pred = calib_pred[,"fit"],
  Calib_Lower = calib_pred[,"lwr"],

# Calculate coverage (% of actual values within prediction intervals)
lm_coverage <- mean(test_data$mpg >= results$LM_Lower & 
                   test_data$mpg <= results$LM_Upper)
calib_coverage <- mean(test_data$mpg >= results$Calib_Lower & 
                      test_data$mpg <= results$Calib_Upper)

print(sprintf("\nPrediction interval coverage:"))
## [1] "\nPrediction interval coverage:"
print(sprintf("Linear Model: %.1f%%", 100 * lm_coverage))
## [1] "Linear Model: 100.0%"
print(sprintf("Calibrated Model: %.1f%%", 100 * calib_coverage))
## [1] "Calibrated Model: 80.0%"
# Set common y-axis limits for both plots
y_limits <- range(c(results$LM_Lower, results$LM_Upper,
                   results$Calib_Lower, results$Calib_Upper))

# Plot prediction intervals
par(mfrow=c(1,2))

# Linear Model Plot
plot(results$Actual, results$LM_Pred, 
     main="Linear Model Predictions",
     xlab="Actual MPG", ylab="Predicted MPG",
     ylim=y_limits)
# Add shaded prediction intervals
x_ordered <- order(results$Actual)
polygon(c(results$Actual[x_ordered], rev(results$Actual[x_ordered])),
